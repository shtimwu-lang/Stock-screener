       function generateStockData(basePrice, volatility, days = 250, startDate = '2024-01-01') {

               const date = new Date(start);
               date.setDate(date.getDate() + i);

               data.push({
                   date: date.toISOString().slice(0, 10),
                   open: +open.toFixed(2),
                   high: +high.toFixed(2),
                   low: +low.toFixed(2),
                   close: +close.toFixed(2),
                   volume: volume,
               });
               price = close;
           }
           return data;
       }

       function calculateMA(data, period) {
           const result = [];
           for (let i = 0; i < data.length; i++) {
               if (i < period - 1) {
                   result.push(null);
               } else {
                   const sum = data.slice(i - period + 1, i + 1).reduce((a, b) => a + b.close, 0);
                   result.push(+(sum / period).toFixed(2));
               }
           }
           return result;
       }
       function calculateMACD(data, fastPeriod = 12, slowPeriod = 26, signalPeriod = 9) {
           const emaFast = [];
           const emaSlow = [];
           const dif = [];
           const dea = [];
           const macd = [];
           let prevEmaFast = data[0].close;
           let prevEmaSlow = data[0].close;
           const kFast = 2 / (fastPeriod + 1);
           const kSlow = 2 / (slowPeriod + 1);
           for (let i = 0; i < data.length; i++) {
               const close = data[i].close;
               if (i === 0) {
                   emaFast.push(close);
                   emaSlow.push(close);
               } else {
                   emaFast.push(close * kFast + emaFast[i - 1] * (1 - kFast));
                   emaSlow.push(close * kSlow + emaSlow[i - 1] * (1 - kSlow));
               }
               const difVal = emaFast[i] - emaSlow[i];
               dif.push(+difVal.toFixed(3));
               if (i === 0) {
                   dea.push(difVal);
               } else {
                   dea.push(+(dea[i - 1] * (signalPeriod - 1) / (signalPeriod + 1) + difVal * 2 / (
                       signalPeriod + 1)).toFixed(3));
               }
               macd.push(+((dif[i] - dea[i]) * 2).toFixed(3));
           }
           return { dif, dea, macd };
       }
       function calculateKDJ(data, period = 9) {
           const k = [];
           const d = [];
           const j = [];
           let prevK = 50;
           let prevD = 50;
           for (let i = 0; i < data.length; i++) {
               if (i < period - 1) {
                   k.push(null);
                   d.push(null);
                   j.push(null);
                   continue;
               }
               const slice = data.slice(i - period + 1, i + 1);
               const highest = Math.max(...slice.map(x => x.high));
               const lowest = Math.min(...slice.map(x => x.low));
               const rsv = ((data[i].close - lowest) / (highest - lowest)) * 100 || 50;
               const kVal = (2 / 3) * prevK + (1 / 3) * rsv;
               const dVal = (2 / 3) * prevD + (1 / 3) * kVal;
               const jVal = 3 * kVal - 2 * dVal;
               k.push(+kVal.toFixed(2));
               d.push(+dVal.toFixed(2));
               j.push(+jVal.toFixed(2));
               prevK = kVal;
               prevD = dVal;
           }
           return { k, d, j };
       }
       function calculateRSI(data, period = 14) {
           const rsi = [];
           let avgGain = 0;
           let avgLoss = 0;
           for (let i = 0; i < data.length; i++) {
               if (i === 0) {
                   rsi.push(null);
                   continue;
               }
               const change = data[i].close - data[i - 1].close;
               const gain = Math.max(change, 0);
               const loss = Math.max(-change, 0);
               if (i <= period) {
                   avgGain += gain;
                   avgLoss += loss;
                   if (i === period) {
                       avgGain /= period;
                       avgLoss /= period;
                   }
               } else {
                   avgGain = (avgGain * (period - 1) + gain) / period;
                   avgLoss = (avgLoss * (period - 1) + loss) / period;
               }
               if (i >= period) {
                   const rs = avgGain / (avgLoss || 1);
                   rsi.push(+(100 - 100 / (1 + rs)).toFixed(2));
               } else {
                   rsi.push(null);
               }
           }
           return rsi;
       }

       function screenStocks(stocks, strategyId) {
           const results = [];
           stocks.forEach(stock => {
               const data = stock.data;
               if (data.length < 30) return;
               let matched = false;
               const closes = data.map(d => d.close);
               const volumes = data.map(d => d.volume);
               switch (strategyId) {
                   case 'ma_golden_cross': {
                       const ma5 = calculateMA(data, 5);
                       const ma20 = calculateMA(data, 20);
                       // 检查最近5根K线是否有上穿
                       for (let i = data.length - 6; i < data.length - 1; i++) {
                           if (ma5[i] !== null && ma20[i] !== null &&
                               ma5[i] <= ma20[i] && ma5[i + 1] > ma20[i + 1]) {
                               matched = true;
                               break;
                           }
                       }
                       break;
                   }
                   case 'volume_breakout': {
                       const last = data[data.length - 1];
                       const avgVol5 = volumes.slice(-6, -1).reduce((a, b) => a + b, 0) / 5;
                       const highest20 = Math.max(...closes.slice(-21, -1));
                       if (last.volume > avgVol5 * 2 && last.close > highest20) {
                           matched = true;
                       }
                       break;
                   }
                   case 'macd_golden_cross': {
                       const { dif, dea } = calculateMACD(data);
                       for (let i = data.length - 6; i < data.length - 1; i++) {
                           if (dif[i] !== undefined && dea[i] !== undefined &&
                               dif[i] <= dea[i] && dif[i + 1] > dea[i + 1]) {
                               matched = true;
                               break;
                           }
                       }
                       break;
                   }
                   case 'kdj_oversold': {
                       const { k, d, j } = calculateKDJ(data);
                       for (let i = data.length - 6; i < data.length - 1; i++) {
                           if (j[i] !== null && j[i] < 20 && j[i + 1] > 20 && k[i + 1] > d[i + 1]) {
                               matched = true;
                               break;
                           }
                       }
                       break;
                   }
                   default:
                       return;
               }
               if (matched) {
                   results.push({ ...stock, reason: strategies.find(s => s.id === strategyId)?.desc || '' });
               }
           });
           return results;
       }

           // 更新K线图
           useEffect(() => {
               if (chartInstance && stock) {
                   const data = stock.data.slice(); // 复制
                   const dates = data.map(d => d.date);
                   const ohlc = data.map(d => [d.open, d.close, d.low, d.high]);
                   const volumes = data.map((d, i) => ({
                       value: d.volume,
                       itemStyle: { color: d.close >= d.open ? '#ef5350' : '#26a69a' }
                   }));
                   const ma5 = calculateMA(data, 5);
                   const ma10 = calculateMA(data, 10);
                   const ma20 = calculateMA(data, 20);
                   const option = {
                       animation: false,
                       legend: {
                           data: ['K线', 'MA5', 'MA10', 'MA20'],

                       tooltip: {
                           trigger: 'axis',
                           axisPointer: { type: 'cross' },

                           textStyle: { color: '#c9d1d9' }
                       },
                       grid: [
                           { left: '8%', right: '5%', top: '30', height: '55%' },
                           { left: '8%', right: '5%', top: '75%', height: '15%' }
                       ],
                       xAxis: [
                           { type: 'category', data: dates, gridIndex: 0, axisLabel: { show: false } },
                           { type: 'category', data: dates, gridIndex: 1, axisLabel: { show: false } }

                       yAxis: [
                           { scale: true, gridIndex: 0, splitLine: { lineStyle: { color: '#21262d' } } },
                           { scale: true, gridIndex: 1, splitLine: { show: false }, axisLabel: { show: false } }

                       dataZoom: [
                           { type: 'inside', xAxisIndex: [0, 1], start: 70, end: 100 },
                           { type: 'slider', xAxisIndex: [0, 1], start: 70, end: 100, bottom: 0 }
                       ],
                       series: [
                           {
                               name: 'K线',
                               type: 'candlestick',
                               data: ohlc,
                               itemStyle: {
                                   color: '#ef5350',
                                   color0: '#26a69a',
                                   borderColor: '#ef5350',
                                   borderColor0: '#26a69a'
                               }
                           },
                           { name: 'MA5', type: 'line', data: ma5, smooth: true, showSymbol: false, lineStyle: { width: 1, color: '#f0b90b' } },
                           { name: 'MA10', type: 'line', data: ma10, smooth: true, showSymbol: false, lineStyle: { width: 1, color: '#58a6ff' } },
                           { name: 'MA20', type: 'line', data: ma20, smooth: true, showSymbol: false, lineStyle: { width: 1, color: '#ec4899' } },

                               name: '成交量',
                               type: 'bar',
                               xAxisIndex: 1,
                               yAxisIndex: 1,
                               data: volumes,
                               barWidth: '60%'
                           }
                       ]
                   };
                   chartInstance.setOption(option, true);
               }
           }, [chartInstance, stock, period]);
           // 更新技术指标副图
           useEffect(() => {
               if (indicatorInstance && stock) {
                   const data = stock.data;
                   const dates = data.map(d => d.date);
                   let option;
                   switch (indicator) {
                       case 'MACD': {
                           const { dif, dea, macd } = calculateMACD(data);
                           option = {
                               animation: false,
                               tooltip: { trigger: 'axis' },
                               legend: { data: ['DIF', 'DEA', 'MACD'], textStyle: { color: '#8b949e' } },
                               grid: { left: '8%', right: '5%', top: '20', bottom: '20' },
                               xAxis: { type: 'category', data: dates, axisLabel: { show: false } },
                               yAxis: { scale: true, splitLine: { lineStyle: { color: '#21262d' } } },
                               series: [
                                   { name: 'DIF', type: 'line', data: dif, showSymbol: false, lineStyle: { color: '#f0b90b' } },
                                   { name: 'DEA', type: 'line', data: dea, showSymbol: false, lineStyle: { color: '#58a6ff' } },
                                   { name: 'MACD', type: 'bar', data: macd, barWidth: '50%', itemStyle: { color: (params) => params.value >= 0 ? '#ef5350' : '#26a69a' } }
                               ]
                           };
                           break;
                       }
                       case 'KDJ': {
                           const { k, d, j } = calculateKDJ(data);
                           option = {
                               animation: false,
                               tooltip: { trigger: 'axis' },
                               legend: { data: ['K', 'D', 'J'], textStyle: { color: '#8b949e' } },
                               grid: { left: '8%', right: '5%', top: '20', bottom: '20' },
                               xAxis: { type: 'category', data: dates, axisLabel: { show: false } },
                               yAxis: { scale: true, splitLine: { lineStyle: { color: '#21262d' } } },
                               series: [
                                   { name: 'K', type: 'line', data: k, showSymbol: false, lineStyle: { color: '#f0b90b' } },

                                   { name: 'J', type: 'line', data: j, showSymbol: false, lineStyle: { color: '#ec4899' } }
                               ]
                           };
                           break;
                       }
                       case 'RSI': {
                           const rsi6 = calculateRSI(data, 6);
                           const rsi12 = calculateRSI(data, 12);
                           const rsi24 = calculateRSI(data, 24);
                           option = {
                               animation: false,
                               tooltip: { trigger: 'axis' },
                               legend: { data: ['RSI6', 'RSI12', 'RSI24'], textStyle: { color: '#8b949e' } },

                                   { name: 'RSI6', type: 'line', data: rsi6, showSymbol: false, lineStyle: { color: '#f0b90b' } },
                                   { name: 'RSI12', type: 'line', data: rsi12, showSymbol: false, lineStyle: { color: '#58a6ff' } },
                                   { name: 'RSI24', type: 'line', data: rsi24, showSymbol: false, lineStyle: { color: '#ec4899' } }
                               ]
                           };
                           break;
                       }
                       default:
                           option = {};
                   }
                   indicatorInstance.setOption(option, true);
               }
           }, [indicatorInstance, stock, indicator]);
           return (
<div>
<div className="chart-container" ref={chartRef}></div>
<div className="indicator-container" ref={indicatorRef}></div>
</div>
           );
       }
       ReactDOM.createRoot(document.getElementById('root')).render(<App />);
